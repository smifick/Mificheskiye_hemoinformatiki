022 | def read_table(path):
023 |     return pd.read_excel(path) if Path(path).suffix.lower() in {'.xlsx', '.xls'} else pd.read_csv(path)

026 | def structure(value):
027 |     if pd.isna(value):
028 |         return None
029 |     mol = Chem.MolFromSmiles(str(value).strip())
030 |     if mol is None or mol.GetNumHeavyAtoms() == 0:
031 |         return None
032 |     # Conservative policy: exclude multi-fragment records; do not guess active component.
033 |     if len(Chem.GetMolFrags(mol)) != 1:
034 |         return None
035 |     return Chem.MolToSmiles(mol, isomericSmiles=True)
036 | 
037 | 
038 | def group_key(smiles):
039 |     mol = Chem.MolFromSmiles(smiles)
040 |     # Generic scaffold groups heteroatom variants and stereoisomers conservatively.
041 |     scaffold = MurckoScaffold.GetScaffoldForMol(mol)
042 |     if scaffold.GetNumAtoms() == 0:
043 |         return 'ACYCLIC'  # all acyclic molecules stay together
044 |     return Chem.MolToSmiles(MurckoScaffold.MakeScaffoldGeneric(scaffold), isomericSmiles=False)

047 | def features(smiles):
048 |     gen = rdFingerprintGenerator.GetMorganGenerator(radius=2, fpSize=2048)
049 |     result = np.zeros((len(smiles), 2048), dtype=np.uint8)
050 |     for i, s in enumerate(smiles):
051 |         DataStructs.ConvertToNumpyArray(gen.GetFingerprint(Chem.MolFromSmiles(s)), result[i])
052 |     return result

055 | def scores(y, pred):
056 |     return {'MAE': float(mean_absolute_error(y, pred)),
057 |             'RMSE': float(np.sqrt(mean_squared_error(y, pred))),
058 |             'R2': float(r2_score(y, pred)) if len(y) > 1 else None}
